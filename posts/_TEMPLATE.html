def consensus_check(validators: list) -> bool:
    """
    Verifica si hay consenso entre validadores.
    Requiere > 2/3 de acuerdo.
    """
    votes = [v.vote() for v in validators]
    threshold = len(validators) * 2 // 3
    return sum(votes) > threshold
